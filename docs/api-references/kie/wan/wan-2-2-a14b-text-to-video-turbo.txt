                      default: '16:9'
                      example: '16:9'

                        Whether to enable prompt expansion. This will use a

                      example: false

                      minimum: 0
                      maximum: 2147483647
                      default: 0
                      example: 0

                      type: string
                      enum:
                        - none
                        - regular
                      default: none
                      example: none
                  required:
                    - prompt
